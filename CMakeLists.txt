set(PICO_SDK_PATH "${CMAKE_CURRENT_LIST_DIR}/deps/pico-sdk" CACHE PATH "Pinned SDK")
set(PICO_NO_PICOTOOL 1 CACHE BOOL "Use local validated RP2040 bin-to-UF2 conversion")
include(${PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(M6_Reborn VERSION 0.1 LANGUAGES C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT PICO_PLATFORM STREQUAL "rp2040" OR NOT PICO_BOARD STREQUAL "pico")
    message(FATAL_ERROR "This project is configured exclusively for original RP2040 Pico")
endif()
set(FREERTOS_KERNEL_PATH "${CMAKE_CURRENT_LIST_DIR}/deps/FreeRTOS-Kernel")
set(FREERTOS_CONFIG_FILE_DIRECTORY "${CMAKE_CURRENT_LIST_DIR}/config")
include(${FREERTOS_KERNEL_PATH}/portable/ThirdParty/GCC/RP2040/FreeRTOS_Kernel_import.cmake)
pico_sdk_init()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(PICOOS_PROFILE 0 CACHE STRING "0=m6-minimal, 1=ST7735S, 2=generic")
set(PICOOS_ROTATION 0 CACHE STRING "0/2=landscape; 1/3=portrait")
set(PICOOS_BUILD_EPOCH 0 CACHE STRING "Build UTC epoch, used as approximate boot time")
configure_file(config/build_info.h.in generated/build_info.h @ONLY)
set(LV_CONF_PATH "${CMAKE_CURRENT_LIST_DIR}/config/lv_conf.h" CACHE STRING "")
set(LV_CONF_BUILD_DISABLE_EXAMPLES ON CACHE BOOL "")
set(LV_CONF_BUILD_DISABLE_DEMOS ON CACHE BOOL "")
set(LV_CONF_BUILD_DISABLE_THORVG_INTERNAL ON CACHE BOOL "")
add_subdirectory(deps/lvgl EXCLUDE_FROM_ALL)
# LVGL deliberately runs in ONE owning UI task. No LVGL calls from other tasks.
target_link_libraries(lvgl PUBLIC pico_stdlib)
add_library(picoos_display STATIC src/tft.c src/ui.c src/cadillac_logo.c)
target_include_directories(picoos_display PUBLIC include config "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(picoos_display PUBLIC pico_stdlib hardware_spi lvgl)
target_compile_definitions(picoos_display PUBLIC PICOOS_PROFILE=${PICOOS_PROFILE} PICOOS_ROTATION=${PICOOS_ROTATION})

function(outputs target)
    pico_enable_stdio_usb(${target} 1)
    pico_enable_stdio_uart(${target} 0)
    pico_set_program_name(${target} ${target})
    pico_set_program_version(${target} "0.1")
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror=return-type)
    target_link_options(${target} PRIVATE "-Wl,--print-memory-usage")
    pico_add_extra_outputs(${target})
    add_custom_command(TARGET ${target} POST_BUILD
        COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_LIST_DIR}/scripts/bin_to_uf2.py
                ${CMAKE_CURRENT_BINARY_DIR}/${target}.bin ${CMAKE_CURRENT_BINARY_DIR}/${target}.uf2
        VERBATIM)
endfunction()
